\begin{frame}{Entscheidungsprobleme}
    \textbf{Definition 2.11.} Das \textit{Entscheidungsproblem}
    \((\Sigma, L)\) für ein Alphabet \(\Sigma\) und eine
    Sprache \(L \subseteq \Sigma^*\) ist, für jedes
    \(x \in \Sigma^*\) zu entscheiden, ob

    \[
        x \in L \text{ oder } x \not\in L.
    \]

    Ein Algorithmus \(A\) \textit{löst} \((\Sigma, L)\), falls für
    alle \(x \in \Sigma^*\) gilt:

    \[
        A(x) = \begin{cases}
            1 & \text{ falls } x \in L, \\
            0 & \text{ falls } x \not\in L.
        \end{cases}
    \]

    Wir sagen, dass \(A\) die Sprache \(L\) \textit{erkennt}.

\end{frame}

\begin{frame}{Endliche Automaten}

\textbf{Definition 3.1.}
Ein (deterministischer) \textit{endlicher Automat (EA)} ist ein
Quintupel \(M = (Q, \Sigma, \delta, q_0, F)\) wobei
\begin{itemize}
    \item \(Q\) eine endliche Menge von \textit{Zuständen} ist,
    \item \(\Sigma\) ein Alphabet \textit{(Eingabealphabet)} ist,
    \item \(\delta \colon Q \times \Sigma \to Q\) die \textit{Übergangsfunktion} ist,
    \item \(q_0 \in Q\) der Anfangszustand ist,
    \item \(F \subseteq Q\) die \textit{akzeptierenden} Zustände sind.
\end{itemize}

\end{frame}

\begin{frame}{Quiz}
    \begin{enumerate}
        \justifying
        \item \textbf{Frage 1.}
            Seien \(L_1 = \{aaab, abb\}^+\) und \(L_2 = \{w \mid |w|_a = |w|_b\}\)
            zwei Sprachen über \(\Sigma = \{a, b\}\). Was ist das koninisch
            kleinste Wort in \(L_1 \cap L_2\)?

        \item \textbf{Frage 2.}
            Sei \(n \in \mathbb N - \{0\}\). Wie viele Wörter der Länge \(n\)
            über dem Alphabet \(\Sigma = \{a, b, c\}\) gibt es, die jeden der drei
            Buchstaben mindestens einmal enthalten?

        \item \textbf{Frage 3.}
            Seien \(L_1\) und \(L_2\) zwei beliebige Sprachen, und seien
            \(A = L_2(L_2 - L_1)\) und \(B = (L_2)^2 - L_2 L_1\).
            Gelten \(A \subseteq B\) und \(B \subseteq A\) immer?

    \end{enumerate}
\end{frame}
